# chameleon_catalog_role_assignment (Resource)

Assigns a principal role to a Polaris catalog role.

## Example Usage

```terraform
resource "chameleon_warehouse" "analytics" {
  name = "analytics"
}

resource "chameleon_principal_role" "data_reader" {
  name = "data_reader"
}

resource "chameleon_catalog_role" "reader" {
  catalog = chameleon_warehouse.analytics.name
  name    = "analytics_reader"
}

# Bind the principal role to the catalog role so principals with
# `data_reader` inherit the catalog-level grants on `analytics`.
resource "chameleon_catalog_role_assignment" "reader_binding" {
  catalog        = chameleon_warehouse.analytics.name
  catalog_role   = chameleon_catalog_role.reader.name
  principal_role = chameleon_principal_role.data_reader.name
}
```
