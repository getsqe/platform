# chameleon_contract (Resource)

Manages an ODCS v3 data contract in the Chameleon platform.

## Example Usage

```terraform
# chameleon_contract registers an ODCS v3 (Open Data Contract Standard) YAML.
# The BFF parses it with open-data-contract-standard, so it must be real ODCS:
# top-level apiVersion/kind/id/version/status and a schema[] whose first entry
# carries physicalName = "<catalog>.<namespace>.<table>".

resource "chameleon_contract" "orders" {
  yaml_content = <<-EOT
    apiVersion: v3.1.0
    kind: DataContract
    id: orders-contract
    name: orders
    version: 1.0.0
    status: active
    domain: analytics

    schema:
      - name: orders
        physicalName: main_warehouse.analytics_db.orders
        physicalType: table
        properties:
          - name: order_id
            logicalType: string
            primaryKey: true
            required: true
          - name: total
            logicalType: number
            required: true
            quality:
              - type: library
                metric: nullValues
                mustBe: 0
        quality:
          - type: library
            metric: rowCount
            mustBeGreaterThan: 0

    team:
      name: analytics
      members:
        - username: orders-team
          role: Owner
  EOT
}

# When the YAML lives in a separate file:
# resource "chameleon_contract" "orders_from_file" {
#   yaml_content = file("${path.module}/orders-contract.yaml")
# }
```
